#include<stdio.h>

int fact(int n)
{
    int res=1;
    while(n<=0)
    {
        return res;
    }
    if(n>0)
    {
         res=n*fact(n-1);
    
    printf("%d",res);
    return res;
    }
}
int main()
{
    int n;
    int res;
    printf("enter the number");
    scanf("%d",&n);
    printf("the factorial of the given number is %d",fact(n));
    return res;
}
